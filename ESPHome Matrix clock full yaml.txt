spi:

display:
  - platform: max7219digit
    id: matrix_display
    cs_pin: GPIO10
    num_chips: 4
    intensity: 4
    rotate_chip: 0
    scroll_enable: false
    update_interval: 50ms

    lambda: |-
      auto now = id(sntp_time).now();
      bool valid = now.is_valid();

      const uint8_t digits[10][3] = {
        {0xFF, 0x81, 0xFF},
        {0x02, 0xFF, 0x00},
        {0xF9, 0x89, 0x8F},
        {0x81, 0x89, 0xFF},
        {0x1F, 0x10, 0xFC},
        {0x8F, 0x89, 0xF9},
        {0xFF, 0x89, 0xF9},
        {0x01, 0xF1, 0x0F},
        {0xFF, 0x89, 0xFF},
        {0x9F, 0x91, 0xFF}
      };

      auto glyph = [&](char c, int col) -> uint8_t {
        if (c >= '0' && c <= '9')
          return digits[c - '0'][col];

        switch (c) {
          case ':':
            return col == 1 ? 0x24 : 0x00;

          // Bodka v dolnej casti znaku
          case '.':
            return col == 1 ? 0x80 : 0x00;

          // Velke pismena
          case 'M':
            return col == 0 ? 0xFE : col == 1 ? 0x0C : 0xFE;

          case 'T':
            return col == 0 ? 0x02 : col == 1 ? 0xFE : 0x02;

          case 'W':
            return col == 0 ? 0xFC : col == 1 ? 0x30 : 0xFC;

          case 'F':
            return col == 0 ? 0xFE : col == 1 ? 0x12 : 0x12;

          case 'S':
            return col == 0 ? 0x4C : col == 1 ? 0x92 : 0x64;

          // Male pismena
          case 'a':
            return col == 0 ? 0x60 : col == 1 ? 0x90 : 0xF0;

          case 'd':
            return col == 0 ? 0xF0 : col == 1 ? 0x90 : 0xFC;

          case 'e':
            return col == 0 ? 0x60 : col == 1 ? 0xB0 : 0x90;

          case 'h':
            return col == 0 ? 0xFE : col == 1 ? 0x10 : 0xE0;

          case 'i':
            return col == 0 ? 0x00 : col == 1 ? 0xF4 : 0x00;

          case 'n':
            return col == 0 ? 0xF8 : col == 1 ? 0x10 : 0xE0;

          case 'o':
            return col == 0 ? 0x60 : col == 1 ? 0x90 : 0x60;

          case 'r':
            return col == 0 ? 0xF0 : col == 1 ? 0x20 : 0x10;

          case 's':
            return col == 0 ? 0x50 : col == 1 ? 0xA8 : 0x48;

          case 't':
            return col == 0 ? 0x10 : col == 1 ? 0xFE : 0x10;

          case 'u':
            return col == 0 ? 0xF0 : col == 1 ? 0x80 : 0xF0;

          case 'y':
            return col == 0 ? 0x88 : col == 1 ? 0x50 : 0x38;

          default:
            return 0;
        }
      };

      int hour = valid ? now.hour : 0;
      int minute = valid ? now.minute : 0;
      int second = valid ? now.second : (millis() / 1000) % 60;

      // Datum vo formate DD.MM.YY
      char date_text[9];
      snprintf(
        date_text,
        sizeof(date_text),
        "%02d.%02d.%02d",
        valid ? now.day_of_month : 1,
        valid ? now.month : 1,
        valid ? (now.year % 100) : 0
      );

      // ESPHome: 1 = nedela, 2 = pondelok, ..., 7 = sobota
      const char *days[] = {
        "Sunday",
        "Monday",
        "Tuesday",
        "Wednesday",
        "Thursday",
        "Friday",
        "Saturday"
      };

      int dow = valid ? now.day_of_week : 1;
      if (dow < 1 || dow > 7)
        dow = 1;

      const char *day_text = days[dow - 1];

      int day_length = 0;
      while (day_text[day_length] != '\0')
        day_length++;

      const int char_width = 4;
      int day_width = day_length * char_width;

      // Obrazovka 1: HH:MM:SS
      auto clock_pixel = [&](int x, int y) -> bool {
        if (x < 0 || x >= 32 || y < 0 || y >= 8)
          return false;

        int positions[6] = {0, 4, 10, 14, 22, 26};
        int values[6] = {
          hour / 10, hour % 10,
          minute / 10, minute % 10,
          second / 10, second % 10
        };

        for (int i = 0; i < 6; i++) {
          int col = x - positions[i];
          if (col >= 0 && col < 3) {
            if (digits[values[i]][col] & (0x80 >> y))
              return true;
          }
        }

        if (second % 2 == 0) {
          if ((x == 8 || x == 19) && (y == 2 || y == 5))
            return true;
        }

        if (!valid && x == 31 && y == 7)
          return true;

        return false;
      };

      // Obrazovka 2: DD.MM.YY
      auto date_pixel = [&](int x, int y) -> bool {
        if (x < 0 || x >= 32 || y < 0 || y >= 8)
          return false;

        int index = x / 4;
        int col = x % 4;

        if (index >= 8 || col == 3)
          return false;

        uint8_t data = glyph(date_text[index], col);
        return (data & (0x80 >> y)) != 0;
      };

      // Obrazovka 3: cely nazov dna v tyzdni
      auto day_pixel = [&](int x, int y, uint32_t local) -> bool {
        if (x < 0 || x >= 32 || y < 0 || y >= 8)
          return false;

        int travel_distance = 32 + day_width;
        int progress = (uint64_t)local * travel_distance / 3000;
        int offset = 32 - progress;
        int virtual_x = x - offset;

        if (virtual_x < 0 || virtual_x >= day_width)
          return false;

        int index = virtual_x / char_width;
        int col = virtual_x % char_width;

        if (index >= day_length || col >= 3)
          return false;

        uint8_t data = glyph(day_text[index], col);
        return (data & (0x80 >> y)) != 0;
      };

      // Cyklus: cas 7 s, datum 2 s, den v tyzdni 3 s
      uint32_t phase = millis() % 65000;

      int screen;
      uint32_t local;

      if (phase < 60000) {
        screen = 0;
        local = phase;
      } else if (phase < 62000) {
        screen = 1;
        local = phase - 60000;
      } else {
        screen = 2;
        local = phase - 62000;
      }

      const int animation_ms = 800;
      bool animating = local < animation_ms;
      int offset = animating
        ? (local * 32 / animation_ms)
        : 32;

      auto screen_pixel = [&](int which, int x, int y, uint32_t t) -> bool {
        if (which == 0)
          return clock_pixel(x, y);
        if (which == 1)
          return date_pixel(x, y);
        return day_pixel(x, y, t);
      };

      it.fill(Color::BLACK);

      for (int x = 0; x < 32; x++) {
        for (int y = 0; y < 8; y++) {
          bool on = false;

          if (!animating) {
            on = screen_pixel(screen, x, y, local);
          } else {
            int previous = (screen + 2) % 3;
            int old_x = x + offset;
            int new_x = x - (32 - offset);

            if (old_x >= 0 && old_x < 32) {
              on = screen_pixel(previous, old_x, y, 5000);
            }

            if (new_x >= 0 && new_x < 32) {
              on = on || screen_pixel(screen, new_x, y, local);
            }
          }

          if (on)
            it.draw_pixel_at(31 - x, y, Color::WHITE);
        }
      }
